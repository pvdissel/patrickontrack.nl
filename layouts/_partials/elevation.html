{{- /* Elevation profile as inline SVG, rendered at build time. dict "profile" [{km, ele}, ...] "dividers" [dict "km" 28 "label" "2"] "label" "aria label" */ -}} {{- $pro := .profile -}} {{- if and $pro (gt (len $pro) 1) -}} {{- $W := 1100.0 }}{{ $H := 170.0 }}{{ $L := 48.0 }}{{ $R := 14.0 }}{{ $T := 14.0 }}{{ $B := 28.0 -}} {{- $pw := sub $W $L $R }}{{ $ph := sub $H $T $B -}} {{- $km := float (index $pro (sub (len $pro) 1)).km -}} {{- if le $km 0.0 }}{{ $km = 1.0 }}{{ end -}} {{- $lo := 99999.0 }}{{ $hi := -99999.0 -}} {{- range $pro }}{{ $e := float .ele }}{{ if lt $e $lo }}{{ $lo = $e }}{{ end }}{{ if gt $e $hi }}{{ $hi = $e }}{{ end }}{{ end -}} {{- $range := sub $hi $lo -}} {{- $step := 50.0 }}{{ if gt $range 400.0 }}{{ $step = 100.0 }}{{ end }}{{ if gt $range 1200.0 }}{{ $step = 250.0 }}{{ end }}{{ if lt $range 60.0 }}{{ $step = 20.0 }}{{ end -}} {{- $y0 := mul (math.Floor (div $lo $step)) $step -}} {{- $y1 := mul (math.Ceil (div $hi $step)) $step -}} {{- if le $y1 $y0 }}{{ $y1 = add $y0 $step }}{{ end -}} {{- $line := "" -}} {{- range $i, $p := $pro -}} {{- $x := add $L (mul $pw (div (float $p.km) $km)) -}} {{- $y := add $T (mul $ph (sub 1 (div (sub (float $p.ele) $y0) (sub $y1 $y0)))) -}} {{- $line = printf "%s%s%.1f %.1f" $line (cond (eq $i 0) "M" " L") $x $y -}} {{- end -}} {{- $area := printf "%s L%.1f %.1f L%.1f %.1f Z" $line (add $L $pw) (add $T $ph) $L (add $T $ph) -}} {{- $kmStep := 1.0 -}} {{- range slice 2.0 5.0 10.0 20.0 50.0 }}{{ if gt (div $km $kmStep) 10.0 }}{{ $kmStep = . }}{{ end }}{{ end -}} {{- end -}}
