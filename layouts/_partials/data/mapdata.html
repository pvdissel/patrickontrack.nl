{{- /* Build Leaflet layers + markers. dict "trail" trail page or false "stages" optional slice of stage pages to draw (default: all stages of trail) "highlight" optional dict {stagePath: true}; others drawn faded. Without it stages alternate colours. "tracks" slice of story pages whose GPX track is drawn as "mine" "dayShades" true: alternate two shades per track (trip pages), tooltips show the date "stays" slice of stay pages (with lat/lon) drawn as markers */ -}} {{- $layers := slice -}}{{- $markers := slice -}} {{- $stages := .stages | default slice -}} {{- if and (not $stages) .trail }}{{ $stages = partialCached "data/trail-stages.html" .trail .trail.Path }}{{ end -}} {{- $hl := .highlight -}} {{- $n := len $stages -}} {{- range $i, $s := $stages -}} {{- $st := partialCached "data/stats.html" $s $s.Path -}} {{- with $st.gpx -}} {{- $kind := cond (eq (mod $i 2) 0) "stage-a" "stage-b" -}} {{- if $hl }}{{ $kind = cond (isset $hl $s.Path) "official" "faded" }}{{ end -}} {{- $layers = $layers | append (dict "kind" $kind "points" .points "name" (T "map_stage_label" (dict "N" $s.Params.number "Title" $s.Title)) "url" $s.RelPermalink) -}} {{- if eq $i 0 -}} {{- $markers = $markers | append (dict "lat" .first.lat "lon" .first.lng "kind" "end" "label" (T "map_start" ($s.Params.from | default $s.Title))) -}} {{- else -}} {{- $markers = $markers | append (dict "lat" .first.lat "lon" .first.lng "kind" "stage" "label" (string $s.Params.number) "popup" (T "map_stage_label" (dict "N" $s.Params.number "Title" $s.Title)) "url" $s.RelPermalink) -}} {{- end -}} {{- if eq $i (sub $n 1) -}} {{- $markers = $markers | append (dict "lat" .last.lat "lon" .last.lng "kind" "end" "label" (T "map_finish" ($s.Params.to | default $s.Title))) -}} {{- end -}} {{- end -}} {{- end -}} {{- $shades := .dayShades -}} {{- $n := 0 -}} {{- range $t := .tracks -}} {{- with $t.Params.track -}} {{- $g := partialCached "gpx/parse.html" . . -}} {{- if $g.ok -}} {{- $kind := "mine" }}{{ if and $shades (eq (mod $n 2) 1) }}{{ $kind = "mine-b" }}{{ end -}} {{- $name := cond $shades (printf "%s · %s" ($t.Date | time.Format "2 Jan") $t.Title) (T "map_my_track" $t.Title) -}} {{- $layers = $layers | append (dict "kind" $kind "points" $g.points "name" $name "url" $t.RelPermalink) -}} {{- $n = add $n 1 -}} {{- end -}} {{- end -}} {{- end -}} {{- range .stays -}} {{- if and .Params.lat .Params.lon -}} {{- $markers = $markers | append (dict "lat" (float .Params.lat) "lon" (float .Params.lon) "kind" "stay" "label" .Title "popup" (printf "%s · %s" .Title (partial "label.html" (dict "prefix" "type" "value" (.Params.stay_type | default "Stay")))) "url" .RelPermalink) -}} {{- end -}} {{- end -}} {{- return dict "layers" $layers "markers" $markers -}}
