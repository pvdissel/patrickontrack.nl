$diff := sub $ele $ref -}} {{- if gt $diff 4.0 }}{{ $up = add $up $diff }}{{ $ref = $ele }}{{ end -}} {{- if lt $diff -4.0 }}{{ $down = sub $down $diff }}{{ $ref = $ele }}{{ end -}} {{- if lt $ele $min }}{{ $min = $ele }}{{ end -}} {{- if gt $ele $max }}{{ $max = $ele }}{{ end -}} {{- end -}} {{- $pts = $pts | append (dict "lat" $lat "lng" $lon "ele" $ele "km" $dist) -}} {{- $plat = $lat }}{{ }}{{ $end = time.AsTime . }}{{ end -}} {{- $f := index $pts 0 }}{{ $l := index $pts (sub $m 1) -}} {{- $out = dict "ok" true "sample" (strings.Contains $path "/samples/") "distance" $dist "ascent" $up
